```python
async def _fetch_recent_penalties(session_key: int) -> list[dict]:
    """Return penalty messages from OpenF1 /v1/race_control (category=Other)."""
    try:
        url = (
            f"https://api.openf1.org/v1/race_control"
            f"?session_key={session_key}&category=Other"
        )
        async with httpx.AsyncClient(timeout=5.0) as client:
            resp = await client.get(url)
            resp.raise_for_status()
            messages = resp.json()
        out: list[dict] = []
        for m in messages or []:
            text = (m.get("message") or "").upper()
            if "PENALTY" in text:
                out.append({
                    "driver_number": m.get("driver_number"),
                    "message": m.get("message"),
                    "date": m.get("date"),
                })
        return out
    except Exception as e:
        logger.warning("push.race_control_fetch_error", error=str(e))
        return []


async def _fetch_fastest_lap(session_key: int) -> tuple[int, float, int] | None:
    """Return (driver_number, lap_duration, lap_number) for the session-min lap.

    Returns None if /v1/laps is empty or the request fails.  Caller compares
    the returned lap_duration against ``_session_min_lap_seconds`` to decide
    whether to fire an event.
    """
    try:
        url = f"https://api.openf1.org/v1/laps?session_key={session_key}"
        async with httpx.AsyncClient(timeout=5.0) as client:
            resp = await client.get(url)
            resp.raise_for_status()
            laps = resp.json()
        if not laps:
            return None
        valid = [l for l in laps if isinstance(l.get("lap_duration"), (int, float)) and l["lap_duration"] > 0]
        if not valid:
            return None
        best = min(valid, key=lambda l: l["lap_duration"])
        return (
            int(best.get("driver_number") or 0),
            float(best["lap_duration"]),
            int(best.get("lap_number") or 0),
        )
    except Exception as e:
        logger.warning("push.laps_fetch_error", error=str(e))
        return None
```

```python
import asyncio
import time
from typing import Mapping

from aioapns import NotificationRequest, PushType
from app.config import (
    APNS_USE_SANDBOX,
    PUSH_COOLDOWN_FASTEST_LAP,
    PUSH_COOLDOWN_PENALTY,
    PUSH_COOLDOWN_POSITION,
    PUSH_COOLDOWN_SAFETY_CAR,
)
from app.data.device_tokens import (
    DeviceToken,
    deactivate_token,
    tokens_for_event_type,
)
from app.services.apns_client import get_apns


# Driver-number → display name lookup will live in app/api/circuits.py for the
# next phase if needed; for now we use the driver code/string passed in event.
EVENT_TITLES: dict[str, str] = {
    "position_change": "Position change",
    "safety_car": "Safety Car",
    "red_flag": "Red Flag",
    "penalty": "Penalty",
    "fastest_lap": "Fastest Lap",
}


def _cooldown_for(event_type: str) -> int:
    if event_type in {"safety_car", "red_flag"}:
        return PUSH_COOLDOWN_SAFETY_CAR
    if event_type == "position_change":
        return PUSH_COOLDOWN_POSITION
    if event_type == "fastest_lap":
        return PUSH_COOLDOWN_FASTEST_LAP
    if event_type == "penalty":
        return PUSH_COOLDOWN_PENALTY
    return 30


# Per-(room, event_type) cooldown timestamps.  Single-task-per-room
# WebSocket loop means no lock is required (asyncio is cooperative).
_push_cooldown_state: dict[tuple[str, str], float] = {}


def should_push(room: str, event_type: str) -> bool:
    """Return True if a push for (room, event_type) is allowed right now."""
    key = (room, event_type)
    cd = _cooldown_for(event_type)
    last = _push_cooldown_state.get(key, 0.0)
    now = time.time()
    if now - last >= cd:
        _push_cooldown_state[key] = now
        return True
    return False


def _build_title(event: Mapping[str, object]) -> str:
    """Return e.g. 'Verstappen: Moved to P2' or 'Penalty: …' if driver absent."""
    event_type = str(event.get("type") or "")
    summary = str(event.get("summary") or EVENT_TITLES.get(event_type, "Live event"))
    driver = event.get("driver")
    if driver:
        return f"{driver}: {summary}"[:64]
    base = EVENT_TITLES.get(event_type, "Live event")
    return f"{base}: {summary}"[:64] if summary != EVENT_TITLES.get(event_type) else base


async def dispatch_push_for_event(
    room: str,
    event: Mapping[str, object],
    commentary_text: str,
    race_name: str,
    environment: str | None = None,
) -> None:
    """Send a push to every device opted into ``event['type']`` for this env.

    Fire-and-forget by caller (asyncio.create_task in routes.py:live_timing).
    Never raises.
    """
    event_type = str(event.get("type") or "")
    env = environment or ("sandbox" if APNS_USE_SANDBOX else "production")

    if event_type not in EVENT_TITLES:
        # Phase 4 detects pit_stop too; CONTEXT.md excludes it from pushes.
        return

    if not should_push(room, event_type):
        logger.debug("push.cooldown_skip", room=room, event_type=event_type)
        return

    # Lazily acquire a session here — this helper is called from a task that
    # has no FastAPI Depends(get_db) wrapping it.
    from app.data.db import _SessionLocal  # type: ignore[attr-defined]
    if _SessionLocal is None:
        logger.warning("push.db_not_initialized", event_type=event_type)
        return

    async with _SessionLocal() as session:
        try:
            recipients = await tokens_for_event_type(session, event_type, env)
        except Exception as e:
            logger.warning("push.tokens_query_failed", error=str(e))
            return

        if not recipients:
            logger.debug("push.no_recipients", event_type=event_type, env=env)
            return

        title = _build_title(event)
        body = (commentary_text or "").strip()[:200]
        payload = {
            "aps": {
                "alert": {"title": title, "body": body},
                "sound": "default",
                "badge": 1,
                "thread-id": "live-race-events",
            },
            "deep_link": "f1ai://live",
            "event_type": event_type,
            "race": race_name,
        }

        async def _send_one(t: DeviceToken) -> None:
            try:
                req = NotificationRequest(
                    device_token=t.apns_token,
                    message=payload,
                    push_type=PushType.ALERT,
                    time_to_live=300,
                )
                res = await get_apns().send_notification(req)
                if not getattr(res, "is_successful", False):
                    desc = str(getattr(res, "description", ""))
                    if desc in {"Unregistered", "BadDeviceToken"}:
                        await deactivate_token(session, t.device_uuid)
                        logger.info(
                            "push.deactivated_stale_token",
                            device_uuid=t.device_uuid,
                            reason=desc,
                        )
                    else:
                        logger.warning(
                            "push.send_unsuccessful",
                            token_suffix=t.apns_token[-8:],
                            reason=desc,
                        )
            except Exception as e:
                logger.warning(
                    "push.send_failed",
                    token_suffix=t.apns_token[-8:],
                    error=str(e),
                )

        await asyncio.gather(*[_send_one(t) for t in recipients])
        logger.info(
            "push.dispatched",
            event_type=event_type,
            recipients=len(recipients),
            room=room,
        )
```
