"""
Device Token Repository
=======================
SQLAlchemy 2.0 async ORM model + repository functions for the
``device_tokens`` table.  Each row is a single iOS device's APNs token plus
its per-event preference list and the APNs environment its token was
issued by (sandbox vs production — RESEARCH.md §Pitfall 1).

Lifecycle:
  - iOS POSTs to ``/api/push/register`` on every foreground activation.
  - ``upsert_token()`` is idempotent on ``device_uuid`` and re-activates rows.
  - When APNs returns ``Unregistered``/``BadDeviceToken``, the dispatch layer
    calls ``deactivate_token()`` (never DELETE — keeps audit trail).
  - ``tokens_for_event_type()`` is the read path used by the dispatch layer
    in Plan 03.

Concurrency: every public function takes an ``AsyncSession``; sessions are
per-request via ``Depends(get_db)``.  No module-level state is mutated.
"""
from __future__ import annotations

from datetime import datetime, timezone
from typing import Any

import structlog
from pydantic import BaseModel, Field
from sqlalchemy import Boolean, DateTime, JSON, String, func, select
from sqlalchemy.dialects.postgresql import insert as pg_insert
from sqlalchemy.dialects.sqlite import insert as sqlite_insert
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy.orm import Mapped, mapped_column

from app.data.db import Base

logger = structlog.get_logger()


# Canonical event type strings.  MUST match the iOS Set<String>
# in NotificationService.swift (RESEARCH.md §Assumption A10).
EVENT_TYPES: tuple[str, ...] = (
    "overtake",
    "safety_car",
    "red_flag",
    "penalty",
    "fastest_lap",
)


class RegisterTokenPayload(BaseModel):
    """POST /api/push/register body."""

    device_uuid: str = Field(min_length=1, max_length=64)
    apns_token: str = Field(min_length=64, max_length=200, pattern=r"^[a-fA-F0-9]+$")
    environment: str = Field(pattern=r"^(sandbox|production)$")
    bundle_id: str = Field(min_length=1, max_length=128)
    enabled_event_types: list[str] = Field(default_factory=list)


class DeviceToken(Base):
    """Per-device APNs registration row."""

    __tablename__ = "device_tokens"

    device_uuid: Mapped[str] = mapped_column(String(64), primary_key=True)
    apns_token: Mapped[str] = mapped_column(String(256), nullable=False)
    environment: Mapped[str] = mapped_column(String(16), nullable=False)
    enabled_event_types: Mapped[list[str]] = mapped_column(
        JSON, nullable=False, default=list
    )
    bundle_id: Mapped[str] = mapped_column(String(128), nullable=False)
    created_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True), nullable=False, server_default=func.now()
    )
    updated_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True),
        nullable=False,
        server_default=func.now(),
        onupdate=lambda: datetime.now(timezone.utc),
    )
    is_active: Mapped[bool] = mapped_column(Boolean, nullable=False, default=True)


def _to_payload_dict(payload: RegisterTokenPayload | dict[str, Any]) -> dict[str, Any]:
    """Accept either a Pydantic model or a plain dict (for tests)."""
    if isinstance(payload, RegisterTokenPayload):
        return payload.model_dump()
    return dict(payload)


async def upsert_token(
    session: AsyncSession,
    payload: RegisterTokenPayload | dict[str, Any],
) -> DeviceToken:
    """Insert-or-update by device_uuid.  Re-activates deactivated rows."""
    data = _to_payload_dict(payload)
    bind = session.get_bind()
    dialect = bind.dialect.name if bind is not None else "postgresql"

    insert_fn = pg_insert if dialect == "postgresql" else sqlite_insert
    stmt = insert_fn(DeviceToken).values(
        device_uuid=data["device_uuid"],
        apns_token=data["apns_token"],
        environment=data["environment"],
        bundle_id=data["bundle_id"],
        enabled_event_types=list(data.get("enabled_event_types") or []),
    ).on_conflict_do_update(
        index_elements=[DeviceToken.device_uuid],
        set_={
            "apns_token": data["apns_token"],
            "environment": data["environment"],
            "bundle_id": data["bundle_id"],
            "enabled_event_types": list(data.get("enabled_event_types") or []),
            "is_active": True,
            "updated_at": datetime.now(timezone.utc),
        },
    )

    await session.execute(stmt)
    await session.commit()

    result = await session.execute(
        select(DeviceToken).where(DeviceToken.device_uuid == data["device_uuid"])
    )
    row = result.scalar_one()
    logger.info(
        "device_tokens.upsert",
        device_uuid=data["device_uuid"],
        environment=data["environment"],
        token_suffix=data["apns_token"][-8:],
    )
    return row


async def deactivate_token(session: AsyncSession, device_uuid: str) -> None:
    """Mark a row inactive without deleting (audit trail)."""
    result = await session.execute(
        select(DeviceToken).where(DeviceToken.device_uuid == device_uuid)
    )
    row = result.scalar_one_or_none()
    if row is None:
        return
    row.is_active = False
    row.updated_at = datetime.now(timezone.utc)
    await session.commit()
    logger.info("device_tokens.deactivated", device_uuid=device_uuid)


async def tokens_for_event_type(
    session: AsyncSession,
    event_type: str,
    environment: str,
) -> list[DeviceToken]:
    """Return active rows whose preferences include this event type."""
    result = await session.execute(
        select(DeviceToken).where(
            DeviceToken.is_active.is_(True),
            DeviceToken.environment == environment,
        )
    )
    rows = list(result.scalars().all())
    # JSON column membership filtering portably across PG and SQLite:
    return [r for r in rows if event_type in (r.enabled_event_types or [])]


async def count_tokens(session: AsyncSession) -> int:
    """Test helper.  Returns total row count regardless of active state."""
    result = await session.execute(select(func.count()).select_from(DeviceToken))
    return int(result.scalar_one())
```

"""
APNs Client Singleton
=====================
Holds the module-level ``aioapns.APNs`` instance.  Initialized once from
``main.py:lifespan()`` (Plan 05) and reused for every push send.

Cert delivery (RESEARCH.md §Open Question Q2):
  - Local dev: ``APNS_CERT_PATH`` points at a PEM on disk.
  - Render:    ``APNS_CERT_B64`` carries a base64-encoded PEM blob; this
               module materializes it to a temp file at startup.

Sandbox vs production: derived from ``APNS_USE_SANDBOX`` in app.config
(which itself is derived from the existing ``ENVIRONMENT`` env var per
STATE.md decision 01-02).
"""
from __future__ import annotations

import base64
import os
import tempfile

import structlog
from aioapns import APNs

from app.config import (
    APNS_BUNDLE_ID,
    APNS_CERT_B64,
    APNS_CERT_PASSWORD,
    APNS_CERT_PATH,
    APNS_USE_SANDBOX,
)

logger = structlog.get_logger()

_client: APNs | None = None
_materialized_path: str | None = None


def _materialize_cert() -> str:
    """Resolve the cert to a filesystem path aioapns can read."""
    global _materialized_path
    if APNS_CERT_B64:
        if _materialized_path and os.path.exists(_materialized_path):
            return _materialized_path
        try:
            pem_bytes = base64.b64decode(APNS_CERT_B64, validate=True)
        except Exception as e:
            logger.error("apns.cert_b64_decode_failed", error=str(e))
            raise
        fd, path = tempfile.mkstemp(prefix="apns-cert-", suffix=".pem")
        with os.fdopen(fd, "wb") as f:
            f.write(pem_bytes)
        _materialized_path = path
        logger.info("apns.cert_materialized", path=path)
        return path
    return APNS_CERT_PATH


def init_apns() -> None:
    """Initialize the singleton.  Idempotent.  Safe to call from lifespan."""
    global _client
    if _client is not None:
        return
    cert_path = _materialize_cert()
    _client = APNs(
        client_cert=cert_path,
        topic=APNS_BUNDLE_ID,
        use_sandbox=APNS_USE_SANDBOX,
        password=APNS_CERT_PASSWORD,
    )
    logger.info(
        "apns.initialized",
        bundle_id=APNS_BUNDLE_ID,
        sandbox=APNS_USE_SANDBOX,
    )


def get_apns() -> APNs:
    if _client is None:
        raise RuntimeError("init_apns() must be called before get_apns()")
    return _client


async def close_apns() -> None:
    """Future-proof shutdown hook.  aioapns 4.x manages its own lifecycle."""
    global _client, _materialized_path
    _client = None
    if _materialized_path and os.path.exists(_materialized_path):
        try:
            os.unlink(_materialized_path)
        except OSError:
            pass
        _materialized_path = None
    logger.info("apns.closed")
```

import structlog
from fastapi import APIRouter, Depends, status
from sqlalchemy.ext.asyncio import AsyncSession

from app.data.db import get_db
from app.data.device_tokens import RegisterTokenPayload, upsert_token

logger = structlog.get_logger()
router = APIRouter()

async def register_push_token(
    payload: RegisterTokenPayload,
